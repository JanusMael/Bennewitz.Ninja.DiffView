using System;

namespace Sample
{
    public sealed class Greeter
    {
        private readonly string _name;

        public Greeter(string name)
        {
            _name = name;
        }

        public string Greet()
        {
            return "Hello, " + _name + "!";
        }

        public string Farewell()
        {
            return "Goodbye, " + _name + ".";
        }

        private static int Count(string text)
        {
            return text.Length;
        }
    }
}
